export const tokenCounts = countTokens(tokensDocument);
export const escapeRegExp = (value) => value.replace(/[.*+?^${}()|[\[\]\\]/g, '\\$&');
export const generatedCssVariable = (key) => {
  const name = tokensCssRaw.match(new RegExp(`(--az-color-brand-${escapeRegExp(key)}):`))?.[1];
  return name ? `var(${name})` : `var(--az-color-brand-${key})`;
};
export const generatedSassVariable = (key) => tokensScssRaw.match(new RegExp(`(\\$az-color-brand-${escapeRegExp(key)}):`))?.[1] ?? `$az-color-brand-${key}`;
export const brandColors = Object.entries(tokensDocument.az.color.brand).map(([key, token]) => {
  const colorValue = resolveValue(`az.color.brand.${key}`);
  return {
    name: key.replace(/(^|[-_])(\w)/g, (_match, _sep, letter) => ` ${letter.toUpperCase()}`).trim(),
    token: `az.color.brand.${key}`,
    hex: typeof colorValue === 'string' ? colorValue : colorValue.hex,
    cssVar: generatedCssVariable(key),
    sassVar: generatedSassVariable(key),
    sourceFileUrl: githubUrl,
    cssFileUrl: tokensCssGithubUrl,
    scssFileUrl: tokensScssGithubUrl,
    jsFileUrl: tokensVarsJsGithubUrl,
    dtsFileUrl: tokensVarsDtsGithubUrl,
    cmyk: token.$extensions?.cmyk,
    pantone: token.$extensions?.pantone,
  };
});

Arizona's brand palette (`az.color.brand.*`), the raw color values every other color token is built from.

## Semantic tokens

Colors named for their purpose, not their appearance (`az.color.semantic.*`). Each one points at a brand color, and components point at these instead of at brand colors directly, so changing one semantic token updates every component that uses it for that purpose. Open a token to see its alias tree, including which component tokens use it.

<TokenDisplay items={getTokenDisplayItems('az.color.semantic.')} layout="table" />

## Component tokens

Tokens scoped to a single component (`az.component.<component>.*`), so each component's design decisions are explicit and inspectable instead of implicit in its CSS. Structure shared by all of a component's variants sits at its root; each variant (a style like `solid` or `outline`, or a size) holds only what it changes. Colors point at the semantic tokens above; sizes, weights, and opacities point at primitive values.

Expand a component to see its tokens by group, or search across every component by name, type, alias, or value (e.g. `semantic.action.default` finds every component token built on it). Each component's Docs page and the **Tokens** tab on its stories show the same tokens in context.
