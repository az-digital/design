import tokensCssRaw from '../../tokens/dist/tokens.css?raw';
import tokensScssRaw from '../../tokens/dist/tokens.scss?raw';
import tokensVarsJsRaw from '../../tokens/dist/tokens.vars.js?raw';
import tokensVarsDtsRaw from '../../tokens/dist/tokens.vars.d.ts?raw';
import { countTokens, downloadJson, downloadText } from './tokenStats';
import { ColorSwatchGrid } from './ColorSwatchGrid';
import { getTokenDisplayItems, resolveValue } from './resolveToken';
import { TokenDisplay } from './TokenDisplay';

export const githubUrl = 'https://github.com/az-digital/design/blob/create-bootstrap-scss-artifact/packages/tokens/tokens.json';
export const tokensCssGithubUrl = 'https://github.com/az-digital/design/blob/create-bootstrap-scss-artifact/packages/tokens/dist/tokens.css';
export const tokensScssGithubUrl = 'https://github.com/az-digital/design/blob/create-bootstrap-scss-artifact/packages/tokens/dist/tokens.scss';
export const tokensVarsJsGithubUrl = 'https://github.com/az-digital/design/blob/create-bootstrap-scss-artifact/packages/tokens/dist/tokens.vars.js';
export const tokensVarsDtsGithubUrl = 'https://github.com/az-digital/design/blob/create-bootstrap-scss-artifact/packages/tokens/dist/tokens.vars.d.ts';
export const specUrl = 'https://www.designtokens.org/tr/2025.10/color/';
export const tokenCounts = countTokens(tokensDocument);
export const escapeRegExp = (value) => value.replace(/[.*+?^${}()|[\[\]\\]/g, '\\$&');
export const generatedCssVariable = (key) => {
  const name = tokensCssRaw.match(new RegExp(`(--az-color-brand-${escapeRegExp(key)}):`))?.[1];
  return name ? `var(${name})` : `var(--az-color-brand-${key})`;
};
export const generatedSassVariable = (key) => tokensScssRaw.match(new RegExp(`(\\$az-color-brand-${escapeRegExp(key)}):`))?.[1] ?? `$az-color-brand-${key}`;
export const brandColors = Object.entries(tokensDocument.az.color.brand).map(([key, token]) => {
  const colorValue = resolveValue(`az.color.brand.${key}`);
  return {
    name: key.replace(/(^|[-_])(\w)/g, (_match, _sep, letter) => ` ${letter.toUpperCase()}`).trim(),
    token: `az.color.brand.${key}`,
    hex: typeof colorValue === 'string' ? colorValue : colorValue.hex,
    cssVar: generatedCssVariable(key),
    sassVar: generatedSassVariable(key),
    sourceFileUrl: githubUrl,
    cssFileUrl: tokensCssGithubUrl,
    scssFileUrl: tokensScssGithubUrl,
    jsFileUrl: tokensVarsJsGithubUrl,
    dtsFileUrl: tokensVarsDtsGithubUrl,
    cmyk: token.$extensions?.cmyk,
    pantone: token.$extensions?.pantone,
  };
});

## Component tokens

Tokens scoped to a specific component (`az.component.*`), so a component's
actual design decisions are explicit and inspectable instead of implicit in
its CSS. Colors alias to the base/brand tokens above; anything without a
base/semantic tier to alias to yet (padding, font size, border radius, ...)
is hard-coded to the component's real production value instead of blocked
on that tier existing — both kinds get repointed to alias through a
semantic tier later, without changing their names, once one exists. Each
component's own Storybook Docs page also shows a filtered, searchable view
of just its tokens — see [Button's](./?path=/docs/components-button--docs) for an
example.
